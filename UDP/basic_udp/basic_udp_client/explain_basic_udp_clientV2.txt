static int check_port(const char *s) {
    // strtol() would also accept leading whitespace and +, - signs
    if (!isdigit((unsigned char)s[0]))
        return -1;

    char *end;
    errno = 0;
    long v = strtol(s, &end, 10);
    if (errno != 0 || *end != '\0' || v < 1 || v > 65535)
        return -1;
    return (int)v;
}

```c
static int check_IP_version(const char *ip, int port, struct sockaddr_storage *addr, socklen_t *len) {
    struct sockaddr_in  *addr4 = (struct sockaddr_in *)addr;
    struct sockaddr_in6 *addr6 = (struct sockaddr_in6 *)addr;

    memset(addr, 0, sizeof(*addr));
    if (inet_pton(AF_INET, ip, &addr4->sin_addr) == 1) {
        addr4->sin_family = AF_INET;
        addr4->sin_port = htons((uint16_t)port);
        *len = sizeof(*addr4);
        return 0;
    }

    memset(addr, 0, sizeof(*addr));
    if (inet_pton(AF_INET6, ip, &addr6->sin6_addr) == 1) {
        addr6->sin6_family = AF_INET6;
        addr6->sin6_port = htons((uint16_t)port);
        *len = sizeof(*addr6);
        return 0;
    }

    return -1;
}
```
this function serve the idea of handle both IP versions, it gonna detemined which IP had been sent by the user, check if it's IPv4 or IPv6, set the necassary configuration, and check if it's valid IP address in first place.

UDP gives no delivery guarantee, and no connection exists to fail. Without a timeout, if the server never replies, the client's recv() blocks the calling thread indefinitely, with no error and no way to recover. The fix makes recv() give up after 5 seconds and return an error the caller can handle, so the client can report the failure and exit instead of hanging forever.

why I use send() and recv() rather than sendto() and recvfrom(), an unconnected socket using recvfrom() would accept a datagram from **any** sender, and the original code never checked who it came from. connect() makes the kernel discard anything not from the server. Once I made that change, sendto()/recvfrom() no longer made sense, since I'd be passing an address that's already fixed and redundant.
